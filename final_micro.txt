<script>
const directions = ["north","south","east","west"];
const names = {north:"North",south:"South",east:"East",west:"West"};
const sampleData = {north:20,south:10,east:35,west:15};

function getCounts(){
  const counts={};
  for(const d of directions){
    const input=document.getElementById(d);
    const value=Number(input.value);
    if(input.value.trim()==="" || !Number.isInteger(value) || value<0){
      throw new Error("Please enter a valid non-negative whole number for every direction.");
    }
    counts[d]=value;
  }
  return counts;
}

function greenSeconds(vehicleCount){
  // Green signal duration = half of the vehicle count.
  return vehicleCount/2;
}

function formatSeconds(seconds){
  return Number.isInteger(seconds) ? String(seconds) : seconds.toFixed(1);
}

function trafficLevel(count){
  if(count<=20) return {name:"Low",className:"low"};
  if(count<=50) return {name:"Medium",className:"medium"};
  return {name:"High",className:"high"};
}

function setSignal(direction,state){
  const signal=document.getElementById("signal-"+direction);
  signal.querySelectorAll(".light").forEach(light=>light.classList.remove("active"));
  signal.querySelector(".light."+state).classList.add("active");
}

function renderTable(counts){
  const tbody=document.getElementById("signalTable");
  tbody.innerHTML="";
  for(const d of directions){
    const count=counts[d];
    const level=trafficLevel(count);
    const green=formatSeconds(greenSeconds(count));
    const row=document.createElement("tr");
    row.innerHTML=`
      <td>${names[d]}</td>
      <td>${count}</td>
      <td><span class="level ${level.className}">${level.name}</span></td>
      <td class="cycle">Green ${green}s → Yellow 5s → Red</td>
    `;
    tbody.appendChild(row);
  }
}

function analyzeTraffic(){
  let counts;
  try{counts=getCounts();}
  catch(error){document.getElementById("notice").textContent=error.message;return;}

  const total=Object.values(counts).reduce((sum,n)=>sum+n,0);
  const busiest=directions.reduce((best,d)=>counts[d]>counts[best]?d:best,directions[0]);
  const duration=greenSeconds(counts[busiest]);

  document.getElementById("total").textContent=total;
  document.getElementById("busiest").textContent=names[busiest];
  document.getElementById("duration").textContent=formatSeconds(duration)+" sec";
  document.getElementById("highest").textContent=names[busiest]+" - "+counts[busiest]+" vehicles";

  for(const d of directions){
    document.getElementById("count-"+d).textContent=counts[d]+" vehicles";
    setSignal(d,counts[d]===0?"red":(d===busiest?"green":"red"));
  }

  renderTable(counts);
  document.getElementById("statusIntro").textContent=
    "Green duration = half the vehicle count for each direction. Yellow remains 5 seconds, then Red while other directions are served. The busiest direction is "+names[busiest]+".";
  document.getElementById("notice").textContent="Traffic analysis completed successfully.";
}
S
function loadSample(){
  for(const d of directions) document.getElementById(d).value=sampleData[d];
  analyzeTraffic();
}

function resetData(){
  for(const d of directions) document.getElementById(d).value=0;
  analyzeTraffic();
  document.getElementById("notice").textContent="All values have been reset.";
}

analyzeTraffic();
</script>
